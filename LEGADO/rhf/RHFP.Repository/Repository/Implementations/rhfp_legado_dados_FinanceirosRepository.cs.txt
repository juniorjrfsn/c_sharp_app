        private static string NormalizarNomeBusca(string nome)
        {
            if (string.IsNullOrWhiteSpace(nome))
                return string.Empty;

            return string.Join(" ", nome
                .Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => x.Trim())
                .Where(x => !string.IsNullOrWhiteSpace(x)));
        }

        private static string[] ObterTokensPesquisaNome(string nome)
        {
            if (string.IsNullOrWhiteSpace(nome))
                return Array.Empty<string>();

            return nome
                .Trim()
                .Split(new[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => x.Trim())
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .ToArray();
        }

        public List<rhfp_legado_dados_financeirosDTO> GetListaSegurados(int matricula = 0, string cpf = null, string nome = null, string competencia = null, int cod_rubrica = 0)
        {
            try
            {
                var cpfNormalizado = UtilitariosHelper.NormalizarCpf(cpf);
                var cpfComMascara = cpf?.Trim() ?? string.Empty;
                var nomeTrim = nome?.Trim();
                var competenciaTrim = competencia?.Trim();

                var temMatricula = matricula > 0;
                var temCpf = !string.IsNullOrWhiteSpace(cpfNormalizado);
                var temNome = !string.IsNullOrWhiteSpace(nomeTrim);
                var temCompetencia = !string.IsNullOrWhiteSpace(competenciaTrim);

                // Sem nenhum critério de identificação, não retorna nada (evita varrer a tabela inteira)
                if (!temMatricula && !temCpf && !temNome)
                    return new List<rhfp_legado_dados_financeirosDTO>();

                // Restrição opcional por competência/rubrica (continua sendo AND sobre o resultado do OR)
                IQueryable<int> matriculasPermitidas = null;
                if (temCompetencia || cod_rubrica > 0)
                {
                    var financeiros = _context.rhfp_legado_dados_financeiros.AsQueryable();

                    if (temCompetencia)
                        financeiros = financeiros.Where(f => f.fin_competencia_ano_mes == competenciaTrim);

                    if (cod_rubrica > 0)
                        financeiros = financeiros.Where(f => f.rub_codigo == cod_rubrica);

                    matriculasPermitidas = financeiros.Select(f => f.dpe_matricula).Distinct();
                }

                // Cada critério vira uma consulta já projetada no DTO
                IQueryable<rhfp_legado_dados_financeirosDTO> Projetar(IQueryable<rhfp_legado_dados_financeiros> q)
                {
                    if (matriculasPermitidas != null)
                        q = q.Where(d => matriculasPermitidas.Contains(d.dpe_matricula));

                    return q.Select(d => new rhfp_legado_dados_financeirosDTO
                    {
                        dpe_matricula = d.dpe_matricula,
                        dpe_nome_servidor = d.dpe_nome_servidor,
                        dpe_cpf_servidor = d.dpe_cpf_servidor
                    });
                }

                var baseQuery = _context.rhfp_legado_dados_financeiros.AsQueryable();
                var partes = new List<IQueryable<rhfp_legado_dados_financeirosDTO>>();

                // OR 1: matrícula
                if (temMatricula)
                {
                    partes.Add(Projetar(baseQuery.Where(d => d.dpe_matricula == matricula)));
                }

                // OR 2: CPF
                if (temCpf)
                {
                    partes.Add(Projetar(baseQuery.Where(d => d.dpe_cpf_servidor != null && (
                        d.dpe_cpf_servidor == cpfNormalizado ||
                        d.dpe_cpf_servidor == cpfComMascara ||
                        d.dpe_cpf_servidor.Replace(".", "").Replace("-", "").Replace("/", "") == cpfNormalizado
                    ))));
                }

                // OR 3: nome (todos os tokens precisam estar contidos no nome)
                if (temNome)
                {
                    var tokens = ObterTokensPesquisaNome(nomeTrim);
                    if (tokens.Length > 0)
                    {
                        var qNome = baseQuery.Where(d => d.dpe_nome_servidor != null);

                        foreach (var token in tokens)
                        {
                            var tokenLower = token.ToLower();
                            qNome = qNome.Where(d => d.dpe_nome_servidor.Trim().ToLower().Contains(tokenLower));
                        }

                        partes.Add(Projetar(qNome));
                    }
                }

                if (partes.Count == 0)
                    return new List<rhfp_legado_dados_financeirosDTO>();

                // Union = OR entre os critérios + remoção de duplicados (equivale ao GROUP BY do SQL)
                var uniao = partes.Aggregate((a, b) => a.Union(b));

                return uniao
                    .OrderBy(x => x.dpe_nome_servidor)
                    .ThenBy(x => x.dpe_matricula)
                    .ToList();
            }
            catch (Exception ex)
            {
                var detalhe = ex.InnerException?.Message ?? ex.Message;
                throw new Exception($"Erro ao consultar segurados. Detalhe: {detalhe}", ex);
            }
        }

        public List<rhfp_legado_dados_financeirosDTO> GetListaFinanceirosSegurados(int matricula = 0, string cpf = null, string nome = null, string competencia = null, int cod_rubrica = 0)
        {
            try
            {
                var cpfNormalizado = UtilitariosHelper.NormalizarCpf(cpf);
                var cpfComMascara = cpf?.Trim() ?? string.Empty;
                var nomeTrim = nome?.Trim();
                var competenciaTrim = competencia?.Trim();

                var temMatricula = matricula > 0;
                var temCpf = !string.IsNullOrWhiteSpace(cpfNormalizado);
                var temNome = !string.IsNullOrWhiteSpace(nomeTrim);
                var temCompetencia = !string.IsNullOrWhiteSpace(competenciaTrim);

                if (!temMatricula && !temCpf && !temNome)
                    return new List<rhfp_legado_dados_financeirosDTO>();

                IQueryable<rhfp_legado_dados_pessoais> pessoaQuery = _context.rhfp_legado_dados_pessoais;

                if (temMatricula)
                    pessoaQuery = pessoaQuery.Where(d => d.dpe_matricula == matricula);

                if (temCpf)
                {
                    pessoaQuery = pessoaQuery.Where(d => d.dpe_cpf_servidor != null && (
                        d.dpe_cpf_servidor == cpfNormalizado ||
                        d.dpe_cpf_servidor == cpfComMascara ||
                        d.dpe_cpf_servidor.Replace(".", "").Replace("-", "").Replace("/", "") == cpfNormalizado
                    ));
                }

                if (temNome)
                {
                    var nomeBusca = NormalizarNomeBusca(nomeTrim);
                    if (!string.IsNullOrWhiteSpace(nomeBusca))
                    {
                        var nomeBuscaLower = nomeBusca.ToLower();
                        pessoaQuery = pessoaQuery.Where(d => d.dpe_nome_servidor != null &&
                            d.dpe_nome_servidor.Trim().ToLower().StartsWith(nomeBuscaLower));
                    }
                }

                var matriculasSelecionadas = pessoaQuery
                    .Select(d => d.dpe_matricula)
                    .Distinct();

                var query = from lf in _context.rhfp_legado_dados_financeiros
                            join d in pessoaQuery on lf.dpe_matricula equals d.dpe_matricula into pessoaJoin
                            from d in pessoaJoin.DefaultIfEmpty()
                            where matriculasSelecionadas.Contains(lf.dpe_matricula)
                            select new
                            {
                                lf,
                                d
                            };

                if (temCompetencia)
                    query = query.Where(x => x.lf.fin_competencia_ano_mes == competenciaTrim);

                if (cod_rubrica > 0)
                    query = query.Where(x => x.lf.rub_codigo == cod_rubrica);

                var resultado = query
                    .Select(x => new rhfp_legado_dados_financeirosDTO
                    {
                        fin_numero = x.lf.fin_numero,
                        dpe_matricula = x.lf.dpe_matricula,
                        dpe_nome_servidor = x.d != null ? x.d.dpe_nome_servidor : null,
                        dpe_cpf_servidor = x.d != null ? x.d.dpe_cpf_servidor : null,
                        dfu_tp_cargo = x.lf.dfu_tp_cargo,
                        dfu_desc_tp_cargo = x.lf.dfu_desc_tp_cargo,
                        fin_competencia_ano_mes = x.lf.fin_competencia_ano_mes,
                        rub_codigo = x.lf.rub_codigo,
                        rub_descricao = x.lf.rub_descricao,
                        fin_dt_inicio = x.lf.fin_dt_inicio,
                        fin_valor = x.lf.fin_valor,
                        fin_perc_pontos_dia_hora = x.lf.fin_perc_pontos_dia_hora,
                        fin_qtd_urv = x.lf.fin_qtd_urv
                    })
                    .OrderBy(x => x.dpe_nome_servidor)
                    .ThenBy(x => x.dpe_matricula)
                    .ThenBy(x => x.fin_competencia_ano_mes)
                    .ToList();

                return resultado;
            }
            catch (Exception ex)
            {
                var detalhe = ex.InnerException?.Message ?? ex.Message;
                throw new Exception($"Erro ao consultar segurados. Detalhe: {detalhe}", ex);
            }
        }

        public List<rhfp_legado_dados_financeirosDTO> GetFinanceiro(
             int matricula = 0,
             string cpf = null,
             string nome = null,
             string competencia = null,
             int cod_rubrica = 0,
             string dtIni = null,
             string dtFim = null
        )
        {

            try
            {
                var cpfNormalizado = UtilitariosHelper.NormalizarCpf(cpf);
                var nomeTrim = nome?.Trim();
                var dtIniFormatada = UtilitariosHelper.NormalizarCompetencia(dtIni);
                var dtFimFormatada = UtilitariosHelper.NormalizarCompetencia(dtFim);

                var temMatricula = matricula > 0;
                var temCpf = !string.IsNullOrWhiteSpace(cpfNormalizado);
                var temNome = !string.IsNullOrWhiteSpace(nomeTrim);
                var temCompetencia = !string.IsNullOrWhiteSpace(competencia);
                var temDtIni = !string.IsNullOrWhiteSpace(dtIniFormatada);
                var temDtFim = !string.IsNullOrWhiteSpace(dtFimFormatada);

                if (!temMatricula && !temCpf && !temNome && !temCompetencia && !temDtIni && !temDtFim)
                    return new List<rhfp_legado_dados_financeirosDTO>();

                // dp NÃO é mais um join direto por matrícula: rhfp_financeiro tem várias linhas
                // por matrícula (uma por competência/rubrica), então um join só por matrícula
                // multiplicava cada linha de lf por todas as linhas de dp com a mesma matrícula
                // (fan-out), duplicando registros e inflando os totais.
                // Em vez disso, usamos uma subquery correlacionada (FirstOrDefault) que traz
                // no máximo 1 nome/cpf por matrícula.

                var query = from lf in _context.rhfp_legado_dados_financeiros
                            select new
                            {
                                lf,
                                dp = _context.rhfp_legado_dados_pessoais
                                    .Where(d => d.dpe_matricula == lf.dpe_matricula)
                                    .Select(d => new { d.dpe_nome_servidor, d.dpe_cpf_servidor })
                                    .FirstOrDefault()
                            };

                if (temMatricula)
                {
                    query = query.Where(x => x.lf.dpe_matricula == matricula);
                }

                if (temCpf)
                {
                    var cpfRaw = cpfNormalizado;
                    var cpfComMascara = cpf?.Trim() ?? string.Empty;
                    query = query.Where(x => x.dp != null && x.dp.dpe_cpf_servidor != null && (
                        x.dp.dpe_cpf_servidor == cpfRaw ||
                        x.dp.dpe_cpf_servidor == cpfComMascara ||
                        x.dp.dpe_cpf_servidor.Replace(".", "").Replace("-", "").Replace("/", "") == cpfRaw
                    ));
                }

                if (temNome)
                {
                    var nomeBusca = NormalizarNomeBusca(nomeTrim);
                    if (!string.IsNullOrWhiteSpace(nomeBusca))
                    {
                        var nomeBuscaLower = nomeBusca.ToLower();
                        query = query.Where(x => x.dp != null && x.dp.dpe_nome_servidor != null &&
                            x.dp.dpe_nome_servidor.Trim().ToLower().StartsWith(nomeBuscaLower));
                    }
                }

                if (temCompetencia)
                    query = query.Where(x => x.lf.fin_competencia_ano_mes == competencia);

                if (temDtIni)
                    query = query.Where(x => x.lf.fin_competencia_ano_mes != null && string.Compare(x.lf.fin_competencia_ano_mes, dtIniFormatada) >= 0);

                if (temDtFim)
                    query = query.Where(x => x.lf.fin_competencia_ano_mes != null && string.Compare(x.lf.fin_competencia_ano_mes, dtFimFormatada) <= 0);

                if (cod_rubrica > 0)
                    query = query.Where(x => x.lf.rub_codigo == cod_rubrica);

                var dados = query.Select(x => new
                {
                    x.lf.fin_numero,
                    x.lf.dpe_matricula,
                    nome_ = x.dp != null ? x.dp.dpe_nome_servidor : null,
                    cpf_ = x.dp != null ? x.dp.dpe_cpf_servidor : null,
                    x.lf.dfu_tp_cargo,
                    x.lf.dfu_desc_tp_cargo,
                    x.lf.fin_competencia_ano_mes,
                    x.lf.rub_codigo,
                    x.lf.rub_descricao,
                    x.lf.fin_dt_inicio,
                    x.lf.fin_valor,
                    x.lf.fin_perc_pontos_dia_hora,
                    x.lf.fin_qtd_urv
                }).ToList();

                if (dados == null || dados.Count == 0)
                    return new List<rhfp_legado_dados_financeirosDTO>();

                var calculado = dados.Select(r => new
                {
                    r,
                    PROVENTO = (r.rub_codigo > 0 && r.rub_codigo < 100) ? r.fin_valor : 0m,
                    DESCONTO = (r.rub_codigo >= 100 && r.rub_codigo <= 400) ? r.fin_valor : 0m
                }).ToList();

                var totaisPorParticao = calculado
                    .GroupBy(x => new { x.r.dpe_matricula, x.r.fin_competencia_ano_mes, x.r.dfu_tp_cargo })
                    .ToDictionary(g => g.Key, g => new
                    {
                        TOTAL_PROVENTO = g.Sum(x => x.PROVENTO),
                        TOTAL_DESCONTO = g.Sum(x => x.DESCONTO)
                    });

                var resultado = calculado.Select(x =>
                {
                    var chave = new { x.r.dpe_matricula, x.r.fin_competencia_ano_mes, x.r.dfu_tp_cargo };
                    var tot = totaisPorParticao[chave];

                    return new rhfp_legado_dados_financeirosDTO
                    {
                        fin_numero = x.r.fin_numero,
                        dpe_matricula = x.r.dpe_matricula,
                        dpe_cpf_servidor = x.r.cpf_,
                        dpe_nome_servidor = x.r.nome_,
                        dfu_tp_cargo = x.r.dfu_tp_cargo,
                        dfu_desc_tp_cargo = x.r.dfu_desc_tp_cargo, // Assuming dfu_desc_tp_cargo is the same as dfu_tp_cargo for now
                        fin_competencia_ano_mes = x.r.fin_competencia_ano_mes,
                        rub_codigo = x.r.rub_codigo,
                        rub_descricao = x.r.rub_descricao,
                        fin_dt_inicio = x.r.fin_dt_inicio,
                        fin_valor = x.r.fin_valor,
                        fin_perc_pontos_dia_hora = x.r.fin_perc_pontos_dia_hora,
                        fin_qtd_urv = x.r.fin_qtd_urv,
                        PROVENTO = x.PROVENTO,
                        DESCONTO = x.DESCONTO,
                        TOTAL_PROVENTO = tot.TOTAL_PROVENTO,
                        TOTAL_DESCONTO = tot.TOTAL_DESCONTO,
                        LIQUIDO = tot.TOTAL_PROVENTO - tot.TOTAL_DESCONTO
                    };
                })
                .OrderBy(x => x.dpe_matricula)
                .ThenBy(x => x.fin_competencia_ano_mes)
                .ThenBy(x => x.dfu_tp_cargo)
                .ToList();

                return resultado;
            }
            catch (Exception ex)
            {
                var detalhe = ex.InnerException?.Message ?? ex.Message;
                throw new Exception($"Erro ao consultar financeiro. Detalhe: {detalhe}", ex);
            }
        }

        /// <summary>
        /// Consulta de Segurados (agrupado por matricula, nome e CPF) com filtros opcionais.
        /// Retorna uma lista de registros únicos de Segurados, agrupados por dpe_matricula, dpe_nome_servidor e dpe_cpf_servidor.
        /// </summary>
        public List<rhfp_legado_dados_financeirosDTO> GetFinanceiroPorDependente(
            int dpe_matricula = 0,
            string dpe_nome_servidor = null,
            string dpe_cpf_servidor = null
        )
        {
            try
            {
                var cpfNormalizado = UtilitariosHelper.NormalizarCpf(dpe_cpf_servidor);
                var nomeTrim = dpe_nome_servidor?.Trim();

                var temMatricula = dpe_matricula > 0;

                var temNome = !string.IsNullOrWhiteSpace(nomeTrim);

                if (!temMatricula && !temCpf && !temNome)
                    return new List<rhfp_legado_dados_financeirosDTO>();

                IQueryable<rhfp_legado_dados_pessoais> dpQuery = _context.rhfp_legado_dados_pessoais.AsQueryable();

                if (temMatricula)
                {
                    dpQuery = dpQuery.Where(d => d.dpe_matricula == dpe_matricula);
                }

                if (temCpf)
                {
                    var cpfRaw = cpfNormalizado;
                    var cpfComMascara = dpe_cpf_servidor?.Trim() ?? string.Empty;
                    dpQuery = dpQuery.Where(d => d.dpe_cpf_servidor != null && (
                        d.dpe_cpf_servidor == cpfRaw ||
                        d.dpe_cpf_servidor == cpfComMascara ||
                        d.dpe_cpf_servidor.Replace(".", "").Replace("-", "").Replace("/", "") == cpfRaw
                    ));
                }

                if (temNome)
                {
                    var nomeBusca = NormalizarNomeBusca(nomeTrim);
                    if (!string.IsNullOrWhiteSpace(nomeBusca))
                    {
                        var nomeBuscaLower = nomeBusca.ToLower();
                        dpQuery = dpQuery.Where(d => d.dpe_nome_servidor != null &&
                            d.dpe_nome_servidor.Trim().ToLower().StartsWith(nomeBuscaLower));
                    }
                }

                var resultado = dpQuery
                    .Select(d => new rhfp_legado_dados_financeirosDTO
                    {
                        dpe_matricula = d.dpe_matricula,
                        dpe_nome_servidor = d.dpe_nome_servidor,
                        dpe_cpf_servidor = d.dpe_cpf_servidor
                    })
                    .OrderBy(x => x.dpe_nome_servidor)
                    .ToList();

                return resultado;
            }
            catch (Exception ex)
            {
                var detalhe = ex.InnerException?.Message ?? ex.Message;
                throw new Exception($"Erro ao consultar financeiro por dependente. Detalhe: {detalhe}", ex);
            }
        }
