    export function validarFiltroSegurado(): boolean {
        const matricula: string = (($('#matricula').val() as string) || '').trim();
        const nome: string = (($('#nome').val() as string) || '').trim();
        const cpfNumeros: string = (($('#cpf_busca').val() as string) || '').replace(/\D/g, '');

        const erros: string[] = [];

        // Matrícula: apenas números e maior que 0
        if (matricula.length > 0) {
            if (!/^\d+$/.test(matricula)) {
                erros.push('A matrícula deve conter apenas números.');
            } else if (parseInt(matricula, 10) <= 0) {
                erros.push('A matrícula deve ser maior que zero.');
            }
        }

        // Nome: mínimo de 3 caracteres
        if (nome.length > 0 && nome.length < 3) {
            erros.push(`Nome incompleto: faltam ${3 - nome.length} letra(s) para a busca (mínimo 3).`);
        }

        // CPF: exatamente 11 dígitos
        if (cpfNumeros.length > 0 && cpfNumeros.length < 11) {
            erros.push(`CPF incompleto: faltam ${11 - cpfNumeros.length} dígito(s).`);
        }

        const nenhumPreenchido = matricula.length === 0 && nome.length === 0 && cpfNumeros.length === 0;

        if (nenhumPreenchido) {
            AtEvenRel.validoFiltro = false;
            AtEvenRel.msgFiltro = 'Por favor, informe pelo menos um critério de busca (Matrícula, Nome ou CPF).';
        } else if (erros.length > 0) {
            AtEvenRel.validoFiltro = false;
            AtEvenRel.msgFiltro = erros.join('\n');
        } else {
            AtEvenRel.validoFiltro = true;
            AtEvenRel.msgFiltro = '';
        }

        return AtEvenRel.validoFiltro; // mantive o retorno original: true = filtro inválido
    }

    export function inicializarDataTableSegurados(dados: any[]) {

            AtEvenRel.dataTableInstance = null;
        }

        $('#table-lista-segurados tbody').empty();

        let html = '';
        dados.forEach((item: any, index: number) => {
            html += `<tr data-index="${index}">
                <td style="text-align:center;">${item.dep_matricula || ''}</td>
                <td>${item.ate_nome || ''}</td>
                <td style="text-align:center;">${mascararCpf(item.ate_cpf_servidor) || ''}</td>
                <td style="text-align: center;">

                        <i class="fa-solid fa-eye text-info fa-lg"></i>
                    </button> 
                </td>
                <td style="text-align: center;"> 
                    <button type="button" class="btn btn-sm btn-outline-secondary btn-ver-segurado-pdf" data-index="${index}" title="Gerar PDF">
                        <i class="fa-solid fa-file-pdf text-danger fa-lg"></i>
                    </button>
                </td>
            </tr>`;
        });

        $('#table-lista-segurados tbody').html(html);

        // Inicializar DataTable ANTES de adicionar event handlers
        if (AtEvenRel.dataTableInstance) {
            AtEvenRel.dataTableInstance.destroy();
        }

        const $table = $('#table-lista-segurados');
        AtEvenRel.dataTableInstance = $table.DataTable({
            paging: true,
            pageLength: 10,
            lengthMenu: [10, 25, 50, 100],
            destroy: true,
            searching: true,
            ordering: true,
            language: {
                url: 'https://cdn.datatables.net/plug-ins/1.13.6/i18n/pt-BR.json'
            }
        });

        // ====== EVENT HANDLERS (aplicados DEPOIS do DataTable ser inicializado) ======
        
        // Botão Ver HTML - usar delegação no contenedor da tabela
        $table.on('click', 'button.btn-ver-segurado-html', function (e) {
            e.stopPropagation();
            e.preventDefault();
            const index = parseInt($(this).data('index'), 10);
            if (index >= 0 && index < dados.length) {
                carregarAtosDoSeguradoClicado(dados[index]);
            }
        });

        // Botão Gerar PDF - usar delegação no contenedor da tabela
        $table.on('click', 'button.btn-ver-segurado-pdf', function (e) {
            e.stopPropagation();
            e.preventDefault();
            const index = parseInt($(this).data('index'), 10);
            if (index >= 0 && index < dados.length) {
                gerarPdfDoSegurado(dados[index]);
            }
        });
    }

    // Transforma aaaammdd em dd/mm/aaaa
    function formatarDateToBr(data: any): string {
        if (data === null || data === undefined) return '';
        const str = String(data).trim();
        if (str.length < 8) return str;
        const ano = str.substring(0, 4);
        const mes = str.substring(4, 6);
        const dia = str.substring(6, 8);
        return `${dia}/${mes}/${ano}`;
    }

    function dataValida(data: any): boolean {
        const str = String(data ?? '').trim();
        return str !== '' && str !== '0' && str !== '00/00/0000' && str !== '30/12/1899' && str !== '01/01/1900';
    }

    function mascararCpf(cpf: any): string {
        const str = String(cpf ?? '').replace(/\D/g, '');
        if (str.length !== 11) return str;
        return `***.${str.substring(3, 6)}.${str.substring(6, 9)}-**`;
    }

    // Diferente do Financeiro: aqui NÃO há agrupamento em subtabelas (por competência,
    // tipo de cargo etc). O resultado é exibido em uma única tabela, na ordem que já
    // vem do servidor, com cada registro sendo um mini-formulário em uma única TD.
    function formatarValorCampo(valor: any, tipo: 'numero' | 'cpf' | 'data' | 'texto' = 'texto'): string {
        if (valor === null || valor === undefined || valor === '') return '';

        if (tipo === 'numero') return String(valor).trim();
        if (tipo === 'cpf') return mascararCpf(valor);
        if (tipo === 'data') return dataValida(valor) ? formatarDateToBr(valor) : '';

        return String(valor).trim();
    }



    export async function gerarHTML(lista: any[]) {
        if (!Array.isArray(lista)) {
            console.warn('gerarHTML recebeu lista inválida:', lista);
            return;
        }

        const containerElement = document.getElementById('lista-ato-eventos');
        if (!containerElement) {
            console.warn('Elemento #lista-ato-eventos não encontrado.');
            return;
        }

        if (lista.length === 0) {
            containerElement.innerHTML = '<div class="alert alert-info">Nenhum registro de Atos e Eventos encontrado.</div>';
            return;
        }

        // Ensure the template is present in the DOM — prefer an existing element, otherwise fetch the file.
        let templateWrapper = document.getElementById('template-ato-eventos-registro') as HTMLElement | null;
        if (!templateWrapper) {
            try {
                const resp = await fetch('/Content/html/MiniFormAtosEventos.html');
                if (resp.ok) {
                    const html = await resp.text();
                    const tmp = document.createElement('div');
                    tmp.innerHTML = html;
                    // append but keep original hidden template in DOM for future clones
                    document.body.appendChild(tmp);
                    templateWrapper = document.getElementById('template-ato-eventos-registro') as HTMLElement | null;
                }
            } catch (e) {
                console.warn('Erro ao carregar MiniFormAtosEventos.html', e);
            }

            if (!templateWrapper) {
                console.warn('Template #template-ato-eventos-registro não encontrado.');
                return;
            }
        }

        // The actual card element inside the wrapper
        const templateCard = templateWrapper.querySelector('.relatorio-ato-evento') as HTMLElement | null;
        if (!templateCard) {
            console.warn('Elemento .relatorio-ato-evento não encontrado dentro do template.');
            return;
        }

                });

    export function validarBusca(filtros: FiltrosAtosEventos): boolean {
        const cpfLimpo = String(filtros.cpf_busca ?? '').replace(/\D/g, '');
        const cpfInformado = cpfLimpo.length > 0;

        const possuiOutroFiltro = Object.keys(filtros)
            .filter(chave => chave !== 'cpf_busca')
            .some(chave => String((filtros as any)[chave] ?? '').trim().length > 0);

        AtEvenRel.msgFiltro = '';

        if (!cpfInformado && !possuiOutroFiltro) {
            AtEvenRel.validoFiltro = false;
            AtEvenRel.msgFiltro = 'Informe pelo menos um filtro para realizar a busca.';
        } else if (cpfInformado && cpfLimpo.length !== 11) {
            AtEvenRel.validoFiltro = false;
            AtEvenRel.msgFiltro = '🔸 O campo CPF deve conter 11 dígitos.';
        } else {
            AtEvenRel.validoFiltro = true;
        }

        return AtEvenRel.validoFiltro;
    }

    export function carregarAtosEventos(filtros: FiltrosAtosEventos) {
        $('#lista-ato-eventos').empty().html('');
        exibirResultado();

        // mostrar alerta de carregamento enquanto busca os registros
        exibirLoadingConsulta('Atos e Eventos', 'Buscando registros. Por favor, aguarde...');


        var jqxhr = $.post("/AtosEventos/GetAtosEventos", filtros as any, function (data) {
            if (data.sucesso && data.lista && data.lista.length > 0) {
                console.log(data);
                const processedData = processDataForTable(data.lista);
                AtEvenRel.gerarHTML(processedData);
                Swal.close();
            } else {
                exibirAvisoSemDados('Não há registros de Atos e Eventos disponíveis para esta consulta');
            }
        }, "json")
            .fail(function (_XMLHttpRequest_, textStatus, errorThrown) {
                console.log("error");
                console.log(_XMLHttpRequest_); console.log(textStatus); console.log(errorThrown);

                let mensagem = 'Não foi possível consultar os Atos e Eventos.';
                try {
                    const resposta = JSON.parse(_XMLHttpRequest_.responseText);
                    if (resposta && resposta.msg) {
                        mensagem = resposta.msg;
                    }
                } catch (e) {
                    if (_XMLHttpRequest_.status === 401 || _XMLHttpRequest_.status === 403) {
                        mensagem = 'Sua sessão expirou ou você não tem permissão para acessar este recurso.';
                    }
                }

                Swal.fire({
                    icon: 'warning',
                    title: 'Atenção',
                    html: mensagem,
                    confirmButtonText: 'Ok'
                });
                Swal.close();
            });

        return jqxhr;
    }

    export function gerarPDF() {
        const filtros = coletarFiltros();

        if (!AtEvenRel.validarBusca(filtros)) {
            Swal.fire({
                icon: 'warning',
                title: '<span style="color:#045C99;font-size:22px;">Atenção!</span>',
                html: '<label style="color:#045C99;font-size:20px;text-align:left;">' + AtEvenRel.msgFiltro + '<label>',
                footer: ScriptsConfig.footerAlert
            });
            return;
        }


        exibirLoadingConsulta('Atos e Eventos PDF', 'Gerando o relatório em PDF. Por favor, não feche esta janela!');

        $.post('/AtosEventos/GerarPdfAtosEventos', filtros as any, function (data) {
            if (data.sucesso) {
                window.open('/AtosEventos/abrirPdfAtosEventosGerado', 'popup', 'height=1080,width=1024,toolbar=no'); // '_blank' , 'popup', 'height=1080,width=1024,toolbar=no'
                Swal.close();
            } else {
                Swal.fire({
                    icon: 'error',
                    title: 'Oops...',
                    html: data.msg,
                });
            }
        }, 'json')
            .fail(function (_XMLHttpRequest_, textStatus, errorThrown) {
                console.log("error");
                console.log(_XMLHttpRequest_); console.log(textStatus); console.log(errorThrown);

                let mensagem = 'Não foi possível gerar o PDF de Atos e Eventos.';
                try {
                    const resposta = JSON.parse(_XMLHttpRequest_.responseText);
                    if (resposta && resposta.msg) {
                        mensagem = resposta.msg;
                    }
                } catch (e) {
                    if (_XMLHttpRequest_.status === 401 || _XMLHttpRequest_.status === 403) {
                        mensagem = 'Sua sessão expirou ou você não tem permissão para acessar este recurso.';
                    }
                }

                Swal.fire({
                    icon: 'warning',
                    title: 'Atenção',
                    html: mensagem,
                    confirmButtonText: 'Ok'
                });
            })
            .always(function () {
                Swal.close();
            });
    }
